procedure TWebModule1.WebModule1ProjectsAction(Sender: TObject;
  Request: TWebRequest; Response: TWebResponse; var Handled: Boolean);
var
  Q: TFDQuery;    // Objeto para realizar la consulta SQL
  JA: TJSONArray; // Arreglo que contendrá la lista de proyectos en formato JSON
  JO: TJSONObject;  // Objeto individual para cada proyecto
begin
   // Inicialización de los objetos de base de datos y JSON
  Q := TFDQuery.Create(nil);
  JA := TJSONArray.Create;
  try
    // Configuración de la conexión y definición de la consulta SQL
    Q.Connection := FDConnection1;
    Q.SQL.Text :=
      'SELECT PROJ_ID, PROJ_NAME, PROJ_DESC, TEAM_LEADER, PRODUCT FROM PROJECT';
    Q.Open;  // Ejecución de la consulta

    // Recorrido de los registros obtenidos de la base de datos
    while not Q.Eof do
    begin
      // Creamos un objeto JSON por cada fila de la tabla
      JO := TJSONObject.Create;
      JO.AddPair('PROJ_ID', Q.FieldByName('PROJ_ID').AsString);
      JO.AddPair('PROJ_NAME', Q.FieldByName('PROJ_NAME').AsString);
      JO.AddPair('PROJ_DESC', Q.FieldByName('PROJ_DESC').AsString);
      JO.AddPair('TEAM_LEADER', Q.FieldByName('TEAM_LEADER').AsString);
      JO.AddPair('PRODUCT', Q.FieldByName('PRODUCT').AsString);
      // Agregamos el objeto al arreglo principal
      JA.AddElement(JO);
      Q.Next;
    end;

    // Configuración de la respuesta HTTP
    Response.ContentType := 'application/json; charset=utf-8';
    Response.Content := JA.ToJSON;

  finally
    // Liberación de memoria de los objetos locales para evitar fugas (Memory Leaks)
    JA.Free;
    Q.Free;
  end;

  // Indicamos que la petición ha sido gestionada correctamente
  Handled := True;
end;
